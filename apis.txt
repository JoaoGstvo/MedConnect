// GET simples
app.get("/exemplo", (req, res) => {
  res.json({ message: "GET simples" });
});

// GET com query (?nome=Joao&idade=25)
app.get("/exemplo-query", (req, res) => {
  const { nome, idade } = req.query;
  res.json({ message: "GET com query", nome, idade });
});

// GET com params (/exemplo/123)
app.get("/exemplo/:id", (req, res) => {
  const { id } = req.params;
  res.json({ message: "GET com params", id });
});

// GET com params + query (/exemplo/123?ativo=true)
app.get("/exemplo/:id/detalhes", (req, res) => {
  const { id } = req.params;
  const { ativo } = req.query;
  res.json({ message: "GET params + query", id, ativo });
});


------------------------------------------------------------------------

// POST simples (body JSON)
app.post("/exemplo", (req, res) => {
  const dados = req.body;
  res.json({ message: "POST simples", dados });
});

// POST com params (/exemplo/123)
app.post("/exemplo/:id", (req, res) => {
  const { id } = req.params;
  const dados = req.body;
  res.json({ message: "POST com params", id, dados });
});

// POST com query (/exemplo?tipo=novo)
app.post("/exemplo-query", (req, res) => {
  const { tipo } = req.query;
  const dados = req.body;
  res.json({ message: "POST com query", tipo, dados });
});

// POST com params + query + body (/exemplo/123?ativo=true)
app.post("/exemplo/:id/detalhes", (req, res) => {
  const { id } = req.params;
  const { ativo } = req.query;
  const dados = req.body;
  res.json({ message: "POST completo", id, ativo, dados });
});
